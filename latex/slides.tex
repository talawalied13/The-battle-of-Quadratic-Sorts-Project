\section{Bubble Sort}

\begin{frame}{Bubble Sort – Principle}
\begin{itemize}
    \item<1-> Compare adjacent elements
    \item<2-> Swap if they are in the wrong order
    \item<3-> Largest element moves to the end in each pass
    \item<4-> Stop early if no swap occurs
\end{itemize}
\end{frame}


\begin{frame}{Bubble Sort – Step-by-Step}
\setcounter{algorithm}{0}
\begin{algorithm}[H]
\caption{Bubble Sort}
\begin{algorithmic}[1]
\For{$i = 0$ to $n-2$} \pause
    \State swapped $\gets$ false \pause
    \For{$j = 0$ to $n-i-2$} \pause
        \If{$L[j] > L[j+1]$} \pause
            \State swap $L[j], L[j+1]$ \pause
            \State swapped $\gets$ true \pause
        \EndIf
    \EndFor \pause
    \If{swapped = false} \pause
        \State \textbf{break}
    \EndIf
\EndFor
\end{algorithmic}
\end{algorithm}
\end{frame}

\begin{frame}{Bubble Sort – Intuition}
\begin{itemize}
    \item<1-> Think of bubbles rising in water
    \item<2-> Largest values move rightwards
    \item<3-> Sorted region grows from the end
\end{itemize}
\end{frame}

\section{Insertion Sort}

\begin{frame}{Insertion Sort – Principle}
\begin{itemize}
    \item<1-> Consider elements one by one
    \item<2-> Compare with already sorted left part
    \item<3-> Shift elements to the right
    \item<4-> Insert element at correct position
\end{itemize}
\end{frame}


\begin{frame}{Insertion Sort – Step-by-Step}
\setcounter{algorithm}{1}
\begin{algorithm}[H]
\caption{Insertion Sort}
\begin{algorithmic}[1]
\For{$i = 1$ to $n-1$} \pause
    \State key $\gets L[i]$ \pause
    \State $j \gets i-1$ \pause
    \While{$j \geq 0$ and $L[j] > key$} \pause
        \State $L[j+1] \gets L[j]$ \pause
        \State $j \gets j-1$ \pause
    \EndWhile
    \State $L[j+1] \gets key$
\EndFor
\end{algorithmic}
\end{algorithm}
\end{frame}

\section{Part C: Correctness Proof of Insertion Sort}

\begin{frame}{Loop Invariant}
\textbf{Invariant Statement}

\medskip
\emph{At the start of iteration \(i\), the subarray \(L[0 \ldots i-1]\) is sorted.}

\medskip
This invariant captures the idea that Insertion Sort always maintains a
sorted prefix of the array.  
Each iteration extends this sorted part by inserting one new element
into its correct position.
\end{frame}

\begin{frame}{Invariant – Initialization}
At the beginning of the algorithm, we start with \(i = 1\).

\medskip
\begin{itemize}
  \item<1-> The subarray \(L[0]\) contains exactly one element
  \item<2-> A single element is trivially sorted
\end{itemize}

\medskip
Therefore, the invariant holds before the first iteration of the loop.
\end{frame}


\begin{frame}{Invariant – Maintenance}
Assume that at the start of iteration \(i\), the subarray
\(L[0 \ldots i-1]\) is sorted.

\medskip
\begin{itemize}
  \item<1-> The element \(L[i]\), called the key, is compared with elements in the sorted subarray
  \item<2-> Larger elements are shifted one position to the right
  \item<3-> The key is inserted into its correct position
\end{itemize}

\medskip
After this process, the subarray \(L[0 \ldots i]\) is sorted, so the
invariant is preserved.
\end{frame}


\begin{frame}{Invariant – Termination}
The loop terminates when \(i = n\), where \(n\) is the length of the array.

\medskip
\begin{itemize}
  \item<1-> By the invariant, the subarray \(L[0 \ldots n-1]\) is sorted
  \item<2-> This subarray is the entire array
\end{itemize}

\medskip
Therefore, when the algorithm terminates, the entire array is sorted.
This proves the correctness of Insertion Sort.
\end{frame}
